' PROBE_PATIENT_API - Discover ISProfile API for diagnoses, billing, and SQL access
'
' Run this macro inside Profile with a valid PHN.
' Results written to: P:\Merkur\EMR_Research\probe-results.txt
'
' Tests:
'   1. Patient object properties (confirmed and new)
'   2. Patient collections (.Problems, .Diagnoses, .Transactions, etc.)
'   3. Global load functions (LoadPatientProblems, LoadSqlStatement, etc.)
'   4. Find Objects / Stored Query API
'   5. SQL Statement execution

dim outFile
outFile = "P:\Merkur\EMR_Research\probe-results.txt"

dim phn
phn = InputBox("Enter a test patient PHN:", "Probe Patient API")
if phn = "" then
    MsgBox "Cancelled.", 0, "Probe"
else

dim p
set p = FindPatientByNHI(phn)

if p is nothing then
    MsgBox "Patient not found for PHN: " & phn, 0, "Probe"
else

rpt = "PROBE_PATIENT_API - " & Now() & vbCrLf
rpt = rpt & "PHN: " & phn & vbCrLf

' ============================================================
' SECTION 1: Confirmed patient properties
' ============================================================
rpt = rpt & "[1] CONFIRMED PATIENT PROPERTIES" & vbCrLf
rpt = rpt & String(50, "-") & vbCrLf

on error resume next

propVal = "" : propVal = p.ID             : rpt = rpt & "  .ID             = " & IIf(err.number=0, propVal, "ERR:" & err.description) & vbCrLf : err.clear

        set item = col(0) : if err.number <> 0 then err.clear
        if not (item is nothing) then
            dim dv

else
    rpt = rpt & "  LoadSqlStatement('TEST')    = OK (obj returned)" & vbCrLf

on error resume next

set sqlResult = RunSqlStatement("TEST")

    rpt = rpt & "  RunSqlStatement('TEST')     = ERR: " & err.description & vbCrLf : err.clear
else
    rpt = rpt & "  RunSqlStatement('TEST')     = OK" & vbCrLf
end if
on error goto 0

' Try ExecuteSql(sql)
on error resume next
set sqlResult = nothing
set sqlResult = ExecuteSql("SELECT 1 FROM RDB$DATABASE")
if err.number <> 0 then
    rpt = rpt & "  ExecuteSql(str)             = ERR: " & err.description & vbCrLf : err.clear
else
    rpt = rpt & "  ExecuteSql(str)             = OK" & vbCrLf

    rpt = rpt & "  RunSQL(str)                 = OK" & vbCrLf
